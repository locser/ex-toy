# K6 Performance Testing Guide - From Zero to Hero

## 1. Tại sao phải test performance?

Trong thực tế làm việc, tôi đã thấy nhiều ứng dụng crash khi lên production vì:

- Không biết ứng dụng chịu được bao nhiêu user đồng thời
- Database connection pool không đủ
- Memory leak khi có nhiều request
- API response time quá chậm ảnh hưởng UX

**K6 giúp bạn phát hiện những vấn đề này trước khi deploy.**

## 3. Script đầu tiên - Hello World

```javascript
// hello-world.js
import http from "k6/http";
import { check, sleep } from "k6";

export default function () {
  // Gửi GET request
  let response = http.get("https://test-api.k6.io/public/crocodiles/");

  // Kiểm tra response
  check(response, {
    "status is 200": (r) => r.status === 200,
    "response time < 500ms": (r) => r.timings.duration < 500,
  });

  sleep(1); // Nghỉ 1 giây
}
```

Chạy test:

```bash
k6 run hello-world.js
```

## 4. Cấu hình Test Scenarios

### Load Test cơ bản:

```javascript
// load-test.js
import http from "k6/http";
import { check, sleep } from "k6";

export let options = {
  stages: [
    { duration: "2m", target: 100 }, // Ramp up to 100 users
    { duration: "5m", target: 100 }, // Stay at 100 users
    { duration: "2m", target: 0 }, // Ramp down
  ],
  thresholds: {
    http_req_duration: ["p(99)<1500"], // 99% of requests must complete below 1.5s
    http_req_failed: ["rate<0.1"], // Error rate must be below 10%
  },
};

export default function () {
  let response = http.get("https://your-api.com/endpoint");

  check(response, {
    "status is 200": (r) => r.status === 200,
  });

  sleep(1);
}
```

## 5. Test API với Authentication

### JWT Token:

```javascript
// auth-test.js
import http from "k6/http";
import { check } from "k6";

export let options = {
  vus: 50,
  duration: "30s",
};

let token;

export function setup() {
  // Login để lấy token
  let loginResponse = http.post("https://your-api.com/auth/login", {
    username: "testuser",
    password: "testpass",
  });

  token = loginResponse.json("access_token");
  return { token: token };
}

export default function (data) {
  let headers = {
    Authorization: `Bearer ${data.token}`,
    "Content-Type": "application/json",
  };

  let response = http.get("https://your-api.com/protected-endpoint", {
    headers: headers,
  });

  check(response, {
    "status is 200": (r) => r.status === 200,
    "has data": (r) => r.json("data") !== undefined,
  });
}
```

## 6. Test CRUD Operations

```javascript
// crud-test.js
import http from "k6/http";
import { check, group } from "k6";

export let options = {
  vus: 10,
  duration: "2m",
};

const BASE_URL = "https://your-api.com/api";

export default function () {
  let userId;

  group("Create User", () => {
    let payload = JSON.stringify({
      name: `User ${Math.random()}`,
      email: `user${Math.random()}@test.com`,
    });

    let response = http.post(`${BASE_URL}/users`, payload, {
      headers: { "Content-Type": "application/json" },
    });

    check(response, {
      "user created": (r) => r.status === 201,
    });

    userId = response.json("id");
  });

  group("Get User", () => {
    let response = http.get(`${BASE_URL}/users/${userId}`);

    check(response, {
      "user retrieved": (r) => r.status === 200,
      "correct user": (r) => r.json("id") == userId,
    });
  });

  group("Update User", () => {
    let payload = JSON.stringify({
      name: "Updated Name",
    });

    let response = http.put(`${BASE_URL}/users/${userId}`, payload, {
      headers: { "Content-Type": "application/json" },
    });

    check(response, {
      "user updated": (r) => r.status === 200,
    });
  });

  group("Delete User", () => {
    let response = http.del(`${BASE_URL}/users/${userId}`);

    check(response, {
      "user deleted": (r) => r.status === 204,
    });
  });
}
```

## 7. Data-Driven Testing

### Sử dụng CSV file:

```javascript
// data-driven-test.js
import http from "k6/http";
import { check } from "k6";
import { SharedArray } from "k6/data";

// Load data từ CSV
const csvData = new SharedArray("users", function () {
  return open("./test-data.csv")
    .split("\n")
    .slice(1)
    .map((line) => {
      const [name, email, age] = line.split(",");
      return { name, email, age };
    });
});

export let options = {
  vus: 10,
  duration: "1m",
};

export default function () {
  // Chọn random data
  const user = csvData[Math.floor(Math.random() * csvData.length)];

  let payload = JSON.stringify(user);

  let response = http.post("https://your-api.com/users", payload, {
    headers: { "Content-Type": "application/json" },
  });

  check(response, {
    "user created": (r) => r.status === 201,
  });
}
```

test-data.csv:

```csv
name,email,age
John Doe,john@example.com,25
Jane Smith,jane@example.com,30
Bob Johnson,bob@example.com,35
```

## 9. Custom Metrics & Advanced Reporting

```javascript
// advanced-metrics.js
import http from "k6/http";
import { check } from "k6";
import { Counter, Rate, Trend } from "k6/metrics";

// Custom metrics
let errorCounter = new Counter("errors");
let loginRate = new Rate("successful_logins");
let loginDuration = new Trend("login_duration");

export let options = {
  vus: 20,
  duration: "1m",
  thresholds: {
    successful_logins: ["rate>0.9"], // 90% successful logins
    login_duration: ["p(95)<2000"], // 95% login < 2s
    errors: ["count<50"], // < 50 errors total
  },
};

export default function () {
  let startTime = Date.now();

  let response = http.post("https://your-api.com/login", {
    username: "user",
    password: "pass",
  });

  let duration = Date.now() - startTime;

  let success = check(response, {
    "login successful": (r) => r.status === 200,
  });

  if (success) {
    loginRate.add(1);
    loginDuration.add(duration);
  } else {
    loginRate.add(0);
    errorCounter.add(1);
  }
}
```

## 10. Testing Patterns theo Domain

### E-commerce API Testing:

```javascript
// ecommerce-test.js
import http from "k6/http";
import { check, group, sleep } from "k6";

export let options = {
  stages: [
    { duration: "1m", target: 20 }, // Normal load
    { duration: "2m", target: 50 }, // Peak hour
    { duration: "1m", target: 0 }, // Cool down
  ],
};

export default function () {
  group("User Journey - Shopping", () => {
    // 1. Browse products
    let products = http.get("https://shop.com/api/products?limit=20");
    check(products, {
      "products loaded": (r) => r.status === 200,
      "has products": (r) => r.json("data.length") > 0,
    });

    sleep(2); // User browsing time

    // 2. View product detail
    let productId = products.json("data.0.id");
    let product = http.get(`https://shop.com/api/products/${productId}`);
    check(product, {
      "product detail loaded": (r) => r.status === 200,
    });

    sleep(3); // User reading time

    // 3. Add to cart
    let cartResponse = http.post(
      "https://shop.com/api/cart/items",
      JSON.stringify({
        productId: productId,
        quantity: 1,
      }),
      {
        headers: { "Content-Type": "application/json" },
      }
    );

    check(cartResponse, {
      "item added to cart": (r) => r.status === 201,
    });

    sleep(1);

    // 4. Checkout process
    let checkout = http.post(
      "https://shop.com/api/orders",
      JSON.stringify({
        paymentMethod: "credit_card",
        shippingAddress: "Test Address",
      }),
      {
        headers: { "Content-Type": "application/json" },
      }
    );

    check(checkout, {
      "order created": (r) => r.status === 201,
      "order has id": (r) => r.json("orderId") !== undefined,
    });
  });
}
```

        stage('Performance Test') {
            steps {

    }

## 12. Best Practices (Kinh nghiệm thực tế)

### DO:

- **Start small**: Bắt đầu với 1-5 users, tăng dần
- **Test early**: Chạy performance test từ khi có API đầu tiên
- **Monitor resources**: Quan sát CPU, Memory, DB connections
- **Use realistic data**: Dùng data giống production
- **Test different scenarios**: Normal load, peak load, stress test

### DON'T:

- **Don't test in production**: Có thể làm crash hệ thống
- **Don't ignore warm-up**: Hệ thống cần thời gian khởi động
- **Don't test without monitoring**: Phải biết tại sao performance kém
- **Don't forget cleanup**: Clear test data sau khi test

### Test Types:

```javascript
// Smoke Test (10 users, 1 minute)
export let options = {
  vus: 10,
  duration: "1m",
};

// Load Test (Normal load)
export let options = {
  stages: [
    { duration: "5m", target: 100 },
    { duration: "10m", target: 100 },
    { duration: "5m", target: 0 },
  ],
};

// Stress Test (Find breaking point)
export let options = {
  stages: [
    { duration: "2m", target: 100 },
    { duration: "5m", target: 200 },
    { duration: "2m", target: 300 },
    { duration: "5m", target: 400 },
    { duration: "10m", target: 400 },
  ],
};

// Spike Test (Sudden traffic increase)
export let options = {
  stages: [
    { duration: "10s", target: 100 },
    { duration: "1m", target: 100 },
    { duration: "10s", target: 1400 }, // Spike
    { duration: "3m", target: 1400 },
    { duration: "10s", target: 100 },
    { duration: "3m", target: 100 },
  ],
};
```

## 13. Debugging & Troubleshooting

### Log HTTP requests:

```javascript
import http from "k6/http";

export let options = {
  vus: 1,
  duration: "10s",
};

export default function () {
  let response = http.get("https://httpbin.org/get", {
    tags: { name: "debug-request" },
  });

  console.log(`Status: ${response.status}`);
  console.log(`Body: ${response.body}`);
  console.log(`Duration: ${response.timings.duration}ms`);
}
```

### Check response content:

```javascript
check(response, {
  "status is 200": (r) => r.status === 200,
  "body contains expected": (r) => r.body.includes("expected-string"),
  "response time OK": (r) => r.timings.duration < 1000,
});
```

### Test Spring Boot Actuator endpoints:

```javascript
// spring-boot-test.js
import http from "k6/http";
import { check, group } from "k6";

const BASE_URL = "http://localhost:8080";

export let options = {
  vus: 30,
  duration: "2m",
};

export default function () {
  group("Application Health", () => {
    let health = http.get(`${BASE_URL}/actuator/health`);
    check(health, {
      "app is healthy": (r) => r.status === 200 && r.json("status") === "UP",
    });
  });

  group("Business Logic", () => {
    let users = http.get(`${BASE_URL}/api/users`);
    check(users, {
      "users endpoint works": (r) => r.status === 200,
    });

    let orders = http.get(`${BASE_URL}/api/orders`);
    check(orders, {
      "orders endpoint works": (r) => r.status === 200,
    });
  });
}
```

## Bài tập thực hành:

1. **Tạo smoke test** cho REST API của bạn
2. **Implement load test** với 100 concurrent users trong 5 phút
3. **Test CRUD operations** với data-driven approach
4. **Setup CI/CD pipeline** chạy performance test tự động
5. **Create stress test** để tìm breaking point của ứng dụng

Bắt đầu từ script đơn giản rồi từ từ nâng cấp. Quan trọng là hiểu được bottleneck ở đâu để optimize đúng chỗ!
