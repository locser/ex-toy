# K6 Performance Testing Guide - From Zero to Hero

## 1. Tại sao phải test performance?

Trong thực tế làm việc, tôi đã thấy nhiều ứng dụng crash khi lên production vì:

- Không biết ứng dụng chịu được bao nhiêu user đồng thời
- Database connection pool không đủ
- Memory leak khi có nhiều request
- API response time quá chậm ảnh hưởng UX

**K6 giúp bạn phát hiện những vấn đề này trước khi deploy.**

## 6. Test CRUD Operations

```javascript
// crud-test.js
import http from "k6/http";
import { check, group } from "k6";

export let options = {
  vus: 10,
  duration: "2m",
};

const BASE_URL = "https://your-api.com/api";

export default function () {
  let userId;

  group("Create User", () => {
    let payload = JSON.stringify({
      name: `User ${Math.random()}`,
      email: `user${Math.random()}@test.com`,
    });

    let response = http.post(`${BASE_URL}/users`, payload, {
      headers: { "Content-Type": "application/json" },
    });

    check(response, {
      "user created": (r) => r.status === 201,
    });

    userId = response.json("id");
  });

  group("Get User", () => {
    let response = http.get(`${BASE_URL}/users/${userId}`);

    check(response, {
      "user retrieved": (r) => r.status === 200,
      "correct user": (r) => r.json("id") == userId,
    });
  });

  group("Update User", () => {
    let payload = JSON.stringify({
      name: "Updated Name",
    });

    let response = http.put(`${BASE_URL}/users/${userId}`, payload, {
      headers: { "Content-Type": "application/json" },
    });

    check(response, {
      "user updated": (r) => r.status === 200,
    });
  });

  group("Delete User", () => {
    let response = http.del(`${BASE_URL}/users/${userId}`);

    check(response, {
      "user deleted": (r) => r.status === 204,
    });
  });
}
```

## 9. Custom Metrics & Advanced Reporting

```javascript
// advanced-metrics.js
import http from "k6/http";
import { check } from "k6";
import { Counter, Rate, Trend } from "k6/metrics";

// Custom metrics
let errorCounter = new Counter("errors");
let loginRate = new Rate("successful_logins");
let loginDuration = new Trend("login_duration");

export let options = {
  vus: 20,
  duration: "1m",
  thresholds: {
    successful_logins: ["rate>0.9"], // 90% successful logins
    login_duration: ["p(95)<2000"], // 95% login < 2s
    errors: ["count<50"], // < 50 errors total
  },
};

export default function () {
  let startTime = Date.now();

  let response = http.post("https://your-api.com/login", {
    username: "user",
    password: "pass",
  });

  let duration = Date.now() - startTime;

  let success = check(response, {
    "login successful": (r) => r.status === 200,
  });

  if (success) {
    loginRate.add(1);
    loginDuration.add(duration);
  } else {
    loginRate.add(0);
    errorCounter.add(1);
  }
}
```

## 10. Testing Patterns theo Domain

### E-commerce API Testing:

```javascript
// ecommerce-test.js
import http from "k6/http";
import { check, group, sleep } from "k6";

export let options = {
  stages: [
    { duration: "1m", target: 20 }, // Normal load
    { duration: "2m", target: 50 }, // Peak hour
    { duration: "1m", target: 0 }, // Cool down
  ],
};

export default function () {
  group("User Journey - Shopping", () => {
    // 1. Browse products
    let products = http.get("https://shop.com/api/products?limit=20");
    check(products, {
      "products loaded": (r) => r.status === 200,
      "has products": (r) => r.json("data.length") > 0,
    });

    sleep(2); // User browsing time

    // 2. View product detail
    let productId = products.json("data.0.id");
    let product = http.get(`https://shop.com/api/products/${productId}`);
    check(product, {
      "product detail loaded": (r) => r.status === 200,
    });

    sleep(3); // User reading time

    // 3. Add to cart
    let cartResponse = http.post(
      "https://shop.com/api/cart/items",
      JSON.stringify({
        productId: productId,
        quantity: 1,
      }),
      {
        headers: { "Content-Type": "application/json" },
      }
    );

    check(cartResponse, {
      "item added to cart": (r) => r.status === 201,
    });

    sleep(1);

    // 4. Checkout process
    let checkout = http.post(
      "https://shop.com/api/orders",
      JSON.stringify({
        paymentMethod: "credit_card",
        shippingAddress: "Test Address",
      }),
      {
        headers: { "Content-Type": "application/json" },
      }
    );

    check(checkout, {
      "order created": (r) => r.status === 201,
      "order has id": (r) => r.json("orderId") !== undefined,
    });
  });
}
```

## 12. Best Practices (Kinh nghiệm thực tế)

### DO:

- **Start small**: Bắt đầu với 1-5 users, tăng dần
- **Test early**: Chạy performance test từ khi có API đầu tiên
- **Monitor resources**: Quan sát CPU, Memory, DB connections
- **Use realistic data**: Dùng data giống production
- **Test different scenarios**: Normal load, peak load, stress test

### DON'T:

- **Don't test in production**: Có thể làm crash hệ thống
- **Don't ignore warm-up**: Hệ thống cần thời gian khởi động
- **Don't test without monitoring**: Phải biết tại sao performance kém
- **Don't forget cleanup**: Clear test data sau khi test

### Test Types:

```javascript
import http from "k6/http";
```

```javascript
  },
};

const CAMPAIGN_ID = __ENV.CAMPAIGN_ID || "12";

export default function () {
  // Random user ID từ 1-1000

  // Random session ID
  const sessionId = Math.random().toString(36).substring(2, 15);

    "X-User-Id": userId.toString(),

    "Content-Type": "application/json",

  const url = `${BASE_URL}/api/v1/giveaway-campaigns/${CAMPAIGN_ID}/participate?level=1`;

    "status is 200": (r) => r.status === 200,
```

### End-to-End User Journey Test:

```javascript
// user-journey-giveaway.js
import http from "k6/http";
import { check, group, sleep } from "k6";

export let options = {
  vus: 50,
  duration: "5m",
  thresholds: {
    http_req_duration: ["p(95)<1000"],
    http_req_failed: ["rate<0.05"],
  },
};

const BASE_URL = __ENV.BASE_URL || "http://localhost:1122";
const CAMPAIGN_ID = __ENV.CAMPAIGN_ID || "12";

export default function () {
  const userId = Math.floor(Math.random() * 10000) + 1;
  const sessionId = Math.random().toString(36).substring(2, 15);

  const headers = {
    "X-User-Id": userId.toString(),
    Cookie: `JSESSIONID=${sessionId}`,
    "Content-Type": "application/json",
  };

  group("User Journey - Giveaway Participation", () => {
    // 1. Check campaign details
    let campaignResponse = http.get(
      `${BASE_URL}/api/v1/giveaway-campaigns/${CAMPAIGN_ID}`,
      {
        headers: headers,
      }
    );

    check(campaignResponse, {
      "campaign details loaded": (r) => r.status === 200,
      "campaign is active": (r) => r.json("data.isActive") === true,
      "has available toys": (r) => r.json("data.availableToys") > 0,
    });

    sleep(1);

    // 2. Check if user can participate
    let canParticipate = campaignResponse.json("data.canParticipate");

    if (canParticipate === 1) {
      // 3. Participate in giveaway
      let participateResponse = http.post(
        `${BASE_URL}/api/v1/giveaway-campaigns/${CAMPAIGN_ID}/participate?level=2`,
        null,
        { headers: headers }
      );

      check(participateResponse, {
        "participation successful": (r) => r.status === 200,
        "received toy": (r) => r.json("data.toy") !== undefined,
        "participation recorded": (r) =>
          r.json("data.participation_id") !== undefined,
      });

      sleep(2);

      // 4. Check participation history
      let historyResponse = http.get(
        `${BASE_URL}/api/v1/users/giveaway-participations?user_id=${userId}&page=1&limit=10`,
        { headers: headers }
      );

      check(historyResponse, {
        "history loaded": (r) => r.status === 200,
        "has participation record": (r) => r.json("data.list.length") > 0,
      });
    } else {
      console.log(
        `User ${userId} cannot participate in campaign ${CAMPAIGN_ID}`
      );
    }
  });
}
```

### Monitoring & Metrics:

```javascript
// monitoring-giveaway.js
import http from "k6/http";
import { check } from "k6";
import { Counter, Rate, Trend } from "k6/metrics";

// Custom metrics
let participationSuccess = new Rate("participation_success");
let participationDuration = new Trend("participation_duration");
let toyClaimed = new Counter("toys_claimed");
let errors = new Counter("participation_errors");

export let options = {
  vus: 100,
  duration: "3m",
  thresholds: {
    participation_success: ["rate>0.95"], // 95% success rate
    participation_duration: ["p(95)<500"], // 95% < 500ms
    errors: ["count<100"], // < 100 errors total
  },
};

const BASE_URL = __ENV.BASE_URL || "http://localhost:1122";
const CAMPAIGN_ID = __ENV.CAMPAIGN_ID || "12";

export default function () {
  const userId = Math.floor(Math.random() * 10000) + 1;
  const sessionId = Math.random().toString(36).substring(2, 15);

  const headers = {
    "X-User-Id": userId.toString(),
    Cookie: `JSESSIONID=${sessionId}`,
    "Content-Type": "application/json",
  };

  const startTime = Date.now();

  const response = http.post(
    `${BASE_URL}/api/v1/giveaway-campaigns/${CAMPAIGN_ID}/participate?level=2`,
    null,
    { headers: headers }
  );

  const duration = Date.now() - startTime;

  const success = check(response, {
    "status is 200": (r) => r.status === 200,
    "has toy data": (r) => r.json("data.toy") !== undefined,
  });

  if (success) {
    participationSuccess.add(1);
    toyClaimed.add(1);
  } else {
    participationSuccess.add(0);
    errors.add(1);
  }

  participationDuration.add(duration);
}
```
